8.1 แกะ struct กับ enum

จุดบนฟิลด์ใช้ได้ทุกครั้ง แต่บางฟังก์ชันอยากได้ชื่อฟิลด์มาวางใน scope เดียวกับตรรกะ การแกะคือการจับคู่รูปแบบกับค่า แล้วผูกชื่อให้ส่วนที่สนใจ

แกะฟิลด์

struct Student {
    name: String,
    score: i32,
}

fn main() {
    let student = Student {
        name: String::from("Ann"),
        score: 85,
    };
    let Student { name, score } = student;
    println!("{name}");
    println!("{score}");
}
Ann
85

หลังแกะ student ถูกย้ายออกเป็น name กับ score เพราะ String ไม่ใช่ Copy ถ้ายังต้องใช้ก้อนเดิม ให้แกะจาก &student แล้วชื่อที่ได้จะเป็นการยืม

แกะค่าที่ variant ถือไว้

enum Answer {
    Known(i32),
    Missing,
}

fn show(answer: Answer) {
    match answer {
        Answer::Known(n) => println!("{n}"),
        Answer::Missing => println!("missing"),
    }
}

fn main() {
    show(Answer::Known(85));
    show(Answer::Missing);
}
85
missing

Answer::Known(n) ทั้งตรวจว่าเป็นทาง Known และดึงตัวเลขมาชื่อ n ทาง Missing ไม่มีค่าให้ดึง

ลองทำให้พัง

ลืมแขนของ enum rustc บอกชื่อ variant ที่ยังไม่ถูกพูดถึง ต่างจากช่วง i32 ในหน้า 3.3 ที่เหลือเป็นช่วงตัวเลขยาวๆ

enum Grade {
    A,
    B,
    C,
}

fn main() {
    let g = Grade::A;
    match g {
        Grade::A => println!("A"),
    }
}

สรุป

  • แกะ struct ได้ชื่อฟิลด์ และถ้าย้าย String ออก ก้อนเดิมใช้ต่อไม่ได้
  • แขน Variant(n) ดึงค่าที่ variant ถือไว้
  • ขาดแขนของ enum แล้ว rustc บอกชื่อที่ยังไม่ครอบคลุม
